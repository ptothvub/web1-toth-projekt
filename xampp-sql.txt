CREATE DATABASE IF NOT EXISTS `upravljanje_vozilima` DEFAULT CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
USE `upravljanje_vozilima`;

CREATE TABLE IF NOT EXISTS `vozila` (
    `id` INT AUTO_INCREMENT PRIMARY KEY,
    `naziv` VARCHAR(100) NOT NULL,
    `registracija` VARCHAR(20) NOT NULL UNIQUE,
    `godina` INT NOT NULL,
    `kilometraza` INT NOT NULL DEFAULT 0,
    `status` ENUM('Dostupan', 'Nedostupan', 'Na servisu') NOT NULL DEFAULT 'Dostupan'
);

CREATE TABLE IF NOT EXISTS `zaposlenici` (
    `id` INT AUTO_INCREMENT PRIMARY KEY,
    `ime_prezime` VARCHAR(100) NOT NULL,
    `lokacija` VARCHAR(100) NOT NULL,
    `kategorije` VARCHAR(50) NOT NULL,
    `status` ENUM('Dostupan', 'Na putu', 'Godišnji odmor') NOT NULL DEFAULT 'Dostupan'
);

CREATE TABLE IF NOT EXISTS `korisnici` (
    `id` INT AUTO_INCREMENT PRIMARY KEY,
    `zaposlenik_id` INT NULL,
    `korisnicko_ime` VARCHAR(50) NOT NULL UNIQUE,
    `lozinka` VARCHAR(50) NOT NULL,
    `uloga` ENUM('admin', 'zaposlenik') NOT NULL DEFAULT 'zaposlenik',
    FOREIGN KEY (`zaposlenik_id`) REFERENCES `zaposlenici`(`id`) ON DELETE CASCADE
);

CREATE TABLE IF NOT EXISTS `najmovi` (
    `id` INT AUTO_INCREMENT PRIMARY KEY,
    `vozilo_id` INT NOT NULL,
    `zaposlenik_id` INT NOT NULL,
    `datum_pocetka` DATE NOT NULL,
    `datum_povratka` DATE NOT NULL,
    `status` ENUM('Aktivno', 'Vraćeno') DEFAULT 'Aktivno',
    FOREIGN KEY (`vozilo_id`) REFERENCES `vozila`(`id`) ON DELETE CASCADE,
    FOREIGN KEY (`zaposlenik_id`) REFERENCES `zaposlenici`(`id`) ON DELETE CASCADE
);
